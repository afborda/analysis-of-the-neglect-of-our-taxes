```python
    SELECT
        e.id_municipio,

        e.sigla_partido,
        e.cargo,
        e.resultado,
```

## Task 5: Notebook 03 — Bolsa Família + Emprego

**Files:**
- Create: `notebooks/fase1_acre/03_bolsa_familia_emprego_ac.ipynb`

- [ ] **Step 1: Coletar dados sociais e de emprego**

```python
import os
from dotenv import load_dotenv
import basedosdados as bd
import pandas as pd
import plotly.express as px

load_dotenv()
PROJECT_ID = os.getenv("GCP_PROJECT_ID")

df_bf = bd.read_sql("""
    SELECT
        id_municipio,
        ano,
        mes,
        quantidade_familias_beneficiadas_bolsa_familia AS familias_bf,
        valor_pago_bolsa_familia                       AS valor_bf
    FROM `basedosdados.br_mc_bolsa_familia.municipio`
    WHERE sigla_uf = 'AC'
    ORDER BY id_municipio, ano, mes
""", billing_project_id=PROJECT_ID)

df_rais = bd.read_sql("""
    SELECT
        id_municipio,
        ano,
        SUM(quantidade_vinculos_ativos) AS vinculos_clt,
        AVG(valor_remuneracao_media)     AS salario_medio
    FROM `basedosdados.br_me_rais.microdados_vinculos`
    WHERE sigla_uf = 'AC'
    GROUP BY id_municipio, ano
    ORDER BY id_municipio, ano
""", billing_project_id=PROJECT_ID)

df_bf.to_json("data/raw/bolsa_familia_ac.json", orient="records")
df_rais.to_json("data/raw/rais_ac.json", orient="records")
print(f"Bolsa Família: {df_bf.shape[0]} registros")
print(f"RAIS: {df_rais.shape[0]} registros")
```

- [ ] **Step 2: Cruzar com população e calcular percentuais**

```python
df_perfil = pd.read_json("data/processed/perfil_municipios_ac.json")

# Média anual do Bolsa Família
df_bf_anual = df_bf.groupby(["id_municipio", "ano"]).agg(
    familias_bf=("familias_bf", "mean"),
    valor_bf=("valor_bf", "sum")
).reset_index()

# Merge com população para calcular %
df_social = df_bf_anual.merge(
    df_perfil[["id_municipio", "nome", "populacao"]],
    on="id_municipio", how="left"
)
df_social["perc_bf"] = (df_social["familias_bf"] * 3.5 / df_social["populacao"] * 100).round(1)

# Merge com emprego
df_completo = df_social.merge(df_rais, on=["id_municipio", "ano"], how="left")
df_completo["perc_clt"] = (df_completo["vinculos_clt"] / df_completo["populacao"] * 100).round(1)

# Snapshot do ano mais recente
ano_ref = df_completo["ano"].max()
df_atual = df_completo[df_completo["ano"] == ano_ref].copy()

# Scatter: % BF vs % CLT
fig = px.scatter(
    df_atual,
    x="perc_clt",
    y="perc_bf",
    text="nome",
    size="populacao",
    title=f"Bolsa Família vs. Emprego Formal — Acre ({ano_ref})",
    labels={"perc_clt": "% Emprego Formal (CLT)", "perc_bf": "% Famílias c/ Bolsa Família"},
    color="perc_bf",
    color_continuous_scale="RdYlGn_r"
)
fig.write_html("data/processed/03_bf_vs_clt.html")
fig.show()
```

- [ ] **Step 3: Asserções e salvar**

```python
assert df_atual["perc_clt"].between(0, 100).all(), "% CLT fora de 0-100"
```

- [ ] **Step 2: Calcular índices financeiros**

```python
df_perfil = pd.read_json("data/processed/perfil_municipios_ac.json")

df_fin = df_siconfi.copy()
df_fin["divida_sobre_receita"]   = (df_fin["divida_consolidada"] / df_fin["receita_liquida"]).round(2)
df_fin["perc_gasto_pessoal"]     = (df_fin["gasto_pessoal"]  / df_fin["receita_liquida"] * 100).round(1)
```

```python
# scrapers/base.py
import json
import time
import random
from pathlib import Path
from abc import ABC, abstractmethod


class BaseScraper(ABC):
    """Classe base para todos os scrapers — garante cache e rate limiting."""

    def __init__(self, cache_dir: str = "data/raw", espera_min: float = 1.0, espera_max: float = 3.0):
        self.cache_dir = Path(cache_dir)
        self.cache_dir.mkdir(parents=True, exist_ok=True)
        self.espera_min = espera_min
        self.espera_max = espera_max

    def cache_path(self, chave: str) -> Path:
        return self.cache_dir / f"{self.__class__.__name__.lower()}_{chave}.json"

    def carregar_cache(self, chave: str):
        path = self.cache_path(chave)
        if path.exists():
            return json.loads(path.read_text(encoding="utf-8"))
        return None

    def salvar_cache(self, chave: str, dados) -> None:
        self.cache_path(chave).write_text(
            json.dumps(dados, ensure_ascii=False, indent=2),
            encoding="utf-8"
        )

    def esperar(self) -> None:
        time.sleep(random.uniform(self.espera_min, self.espera_max))

    @abstractmethod
    def coletar(self, **kwargs):
        pass
```

```python
# scrapers/tse_ficha_limpa.py
import sys
from playwright.sync_api import sync_playwright, TimeoutError as PlaywrightTimeout
from scrapers.base import BaseScraper


class TSEFichaLimpaScraper(BaseScraper):

    URL_BASE = "https://divulgacandcontas.tse.jus.br/divulga/#/candidato"

    def coletar(self, id_candidato: str, ano_eleicao: int) -> dict:
        chave = f"{id_candidato}_{ano_eleicao}"
        cached = self.carregar_cache(chave)
        if cached:
            return cached

        resultado = {"id_candidato": id_candidato, "ano": ano_eleicao, "situacao": None, "erro": None}

        with sync_playwright() as p:
            browser = p.chromium.launch(headless=True)
            page = browser.new_page()
            try:
                url = f"{self.URL_BASE}/{ano_eleicao}/BR/BR/{id_candidato}"
                page.goto(url, wait_until="networkidle", timeout=30000)
                self.esperar()

                seletor = "span.situacao-candidato, .situacao-elegibilidade"
                page.wait_for_selector(seletor, timeout=10000)
                resultado["situacao"] = page.inner_text(seletor).strip()

            except PlaywrightTimeout:
                resultado["erro"] = "timeout"
            except Exception as e:
                resultado["erro"] = str(e)
            finally:
                browser.close()

        self.salvar_cache(chave, resultado)
        return resultado


if __name__ == "__main__":
    uf = sys.argv[1] if len(sys.argv) > 1 else "AC"
    print(f"Coletando ficha limpa para UF={uf}")
    scraper = TSEFichaLimpaScraper()
    # Exemplo de uso — em produção, passar lista de candidatos eleitos
    print("Use com df_eleicoes para passar a lista de candidatos")
```

```python
def normalizar(serie: pd.Series, inverter: bool = False) -> pd.Series:
    """MinMax normalização para 0-1. inverter=True quando maior = pior."""
    scaler = MinMaxScaler()
    s = serie.fillna(serie.median())
    valores = scaler.fit_transform(s.values.reshape(-1, 1)).flatten()
    return pd.Series(1 - valores if inverter else valores, index=serie.index)


# 1. Saúde Financeira (20%)
fin_ref = df_fin.groupby("id_municipio").last().reset_index()
df_base = df_base.merge(fin_ref[["id_municipio", "divida_sobre_receita", "perc_gasto_pessoal"]], on="id_municipio", how="left")
df_base["dim_saude_fin"] = (
    normalizar(df_base["divida_sobre_receita"], inverter=True) * 0.6 +
    normalizar(df_base["perc_gasto_pessoal"],   inverter=True) * 0.4
)

# 2. Qualidade de Vida / IDH (20%)
df_base["dim_qualidade_vida"] = normalizar(df_base["idhm"])

# 3. Emprego e Autonomia (15%)
soc_ref = df_social.groupby("id_municipio").last().reset_index()
df_base = df_base.merge(soc_ref[["id_municipio", "perc_clt", "salario_medio"]], on="id_municipio", how="left")
```
